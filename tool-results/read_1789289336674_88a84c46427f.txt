     2→
     3→// FMCore ERP — Record Detail Drawer (slide-in panel from right)
     4→// Replaces the View modal with a richer UX: tabs for Details / History / Activity,
     5→// inline workflow actions, inline field editing, and a timeline of status transitions.
     6→import { useEffect, useState, useCallback, useRef } from 'react';
     7→import { createPortal } from 'react-dom';
     8→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter } from '@/components/ui/dialog';
     9→import { recordsApi, masterDataApi, uploadsApi } from '@/lib/erp/api';
    10→import { useErpStore } from '@/lib/erp/store';
    11→import type { Register, RecordData, ColumnDef, ColumnType } from '@/lib/erp/types';
    12→import { FAIcon } from './icon';
    13→import { EmptyStateIllustration } from './empty-state-illustration';
    14→import { ApprovalWorkflow } from './approval-workflow';
    15→import { printRecord } from './print-record';
    16→import { WOStageWorkflow } from './wo-stage-workflow';
    17→import { cn } from '@/lib/utils';
    18→import { formatCell, formatDate, formatTimeAgo, statusVariant, validateRecord, defaultValue, displayColumnName } from '@/lib/erp/utils';
    19→import { toast } from 'sonner';
    20→import { Button } from '@/components/ui/button';
    21→import { Input } from '@/components/ui/input';
    22→import { Textarea } from '@/components/ui/textarea';
    23→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
    24→import {
    25→  X, Printer, Pencil, Workflow as WorkflowIcon, History as HistoryIcon,
    26→  FileText, Activity, Clock, CheckCircle2, ArrowRight, Loader2,
    27→  Check, XCircle, AlertCircle, Star, Download, Link2, Upload,
    28→} from 'lucide-react';
    29→
    30→interface HistoryEntry {
    31→  id: string;
    32→  action: string;
    33→  summary: string;
    34→  userName: string;
    35→  userAvatar: string;
    36→  userRole: string | null;
    37→  statusChange: { from: string; to: string } | null;
    38→  createdAt: string;
    39→}
    40→
    41→interface Props {
    42→  open: boolean;
    43→  register: Register;
    44→  record: RecordData | null;
    45→  company: { name: string; address: string; phone: string; email: string; tax_number: string };
    46→  onClose: () => void;
    47→  onEdit: () => void;
    48→  onRefresh: () => void;
    49→}
    50→
    51→type Tab = 'details' | 'history' | 'related' | 'activity';
    52→
    53→export function RecordDetailDrawer({ open, register, record, company, onClose, onEdit, onRefresh }: Props) {
    54→  const { hasPermission, user, currency } = useErpStore();
    55→  const [tab, setTab] = useState<Tab>('details');
    56→  const [history, setHistory] = useState<HistoryEntry[]>([]);
    57→  // Initialize as true since data loads on mount (component is keyed by record id)
    58→  const [historyLoading, setHistoryLoading] = useState(true);
    59→  const [related, setRelated] = useState<any[]>([]);
    60→  const [relatedLoading, setRelatedLoading] = useState(false);
    61→  const [workflowOpen, setWorkflowOpen] = useState(false);
    62→  const [editMode, setEditMode] = useState(false);
    63→  const [editData, setEditData] = useState<Record<string, any>>({});
    64→  const [errors, setErrors] = useState<Record<string, string>>({});
    65→  const [saving, setSaving] = useState(false);
    66→  const [masterData, setMasterData] = useState<Record<string, string[]>>({});
    67→
    68→  const canEdit = hasPermission(register.code, 'edit');
    69→  const canApprove = hasPermission(register.code, 'approve');
    70→  const hasStatusCol = register.columns.some((c) => c.type === 'status');
    71→
    72→  // Load master data for dropdowns
    73→  useEffect(() => {
    74→    if (!open) return;
    75→    let cancelled = false;
    76→    masterDataApi.list().then((md) => {
    77→      if (!cancelled) setMasterData(md);
    78→    }).catch(() => {});
    79→    return () => { cancelled = true; };
    80→  }, [open]);
    81→
    82→  // Initialize edit data when entering edit mode
    83→  const enterEditMode = useCallback(() => {
    84→    if (!record) return;
    85→    setEditData({ ...record.data });
    86→    setErrors({});
    87→    setEditMode(true);
    88→  }, [record]);
    89→
    90→  const cancelEdit = useCallback(() => {
    91→    setEditMode(false);
    92→    setEditData({});
    93→    setErrors({});
    94→  }, []);
    95→
    96→  const setField = (name: string, value: any) => {
    97→    setEditData((d) => ({ ...d, [name]: value }));
    98→    setErrors((e) => { const n = { ...e }; delete n[name]; return n; });
    99→  };
   100→
   101→  const saveInlineEdit = async () => {
   102→    if (!record || !register) return;
   103→    const { valid, errors: errs } = validateRecord(editData, register.columns);
   104→    if (!valid) {
   105→      setErrors(errs);
   106→      toast.error('Please fix the errors before saving', {
   107→        description: `${Object.keys(errs).length} field(s) need attention`,
   108→      });
   109→      return;
   110→    }
   111→    setSaving(true);
   112→    try {
   113→      await recordsApi.update(register.id, record.id, editData);
   114→      toast.success('Record updated successfully');
   115→      setEditMode(false);
   116→      onRefresh();
   117→    } catch (e: any) {
   118→      toast.error('Failed to save record', { description: e.message });
   119→    } finally {
   120→      setSaving(false);
   121→    }
   122→  };
   123→
   124→  // Load history when drawer opens or record changes
   125→  useEffect(() => {
   126→    if (!open || !record || !register) return;
   127→    let cancelled = false;
   128→    recordsApi.getHistory(register.id, record.id).then((res) => {
   129→      if (cancelled) return;
   130→      setHistory(res.history || []);
   131→    }).catch((e) => {
   132→      if (cancelled) return;
   133→      console.error('Failed to load history', e);
   134→    }).finally(() => {
   135→      if (cancelled) return;
   136→      setHistoryLoading(false);
   137→    });
   138→    return () => { cancelled = true; };
   139→  }, [open, record, register.id]);
   140→
   141→  // Load related records when the Related tab is clicked
   142→  useEffect(() => {
   143→    if (tab !== 'related' || !record) return;
   144→    let cancelled = false;
   145→    setRelatedLoading(true);
   146→    const relatedPromise = register ? recordsApi.getRelated(register.id, record.id) : Promise.resolve({ related: [] });
   147→    relatedPromise.then((res) => {
   148→      if (cancelled) return;
   149→      setRelated(res.related || []);
   150→    }).catch((e) => {
   151→      if (cancelled) return;
   152→      console.error('Failed to load related records', e);
   153→    }).finally(() => {
   154→      if (cancelled) return;
   155→      setRelatedLoading(false);
   156→    });
   157→    return () => { cancelled = true; };
   158→  }, [tab, record, register.id]);
   159→
   160→  // Close on Escape
   161→  useEffect(() => {
   162→    if (!open) return;
   163→    const handler = (e: KeyboardEvent) => {
   164→      if (e.key === 'Escape' && !workflowOpen) {
   165→        if (editMode) {
   166→          cancelEdit();
   167→        } else {
   168→          onClose();
   169→        }
   170→      }
   171→    };
   172→    window.addEventListener('keydown', handler);
   173→    return () => window.removeEventListener('keydown', handler);
   174→  }, [open, workflowOpen, onClose, editMode, cancelEdit]);
   175→
   176→  // Exit edit mode when drawer closes
   177→  useEffect(() => {
   178→    if (!open && editMode) {
   179→      setEditMode(false);
   180→      setEditData({});
   181→      setErrors({});
   182→    }
   183→  }, [open, editMode]);
   184→
   185→  const statusCol = register ? register.columns.find((c) => c.type === 'status') : null;
   186→  const currentStatus = statusCol && record ? String(record.data[statusCol.name] || '—') : null;
   187→
   188→  const handlePrint = () => {
   189→    if (!register || !record) return;
   190→    if (register && record) printRecord(register, record, company, currency);
   191→  };
   192→
   193→  const handleWorkflowClick = () => {
   194→    setWorkflowOpen(true);
   195→  };
   196→
   197→  const handleWorkflowTransition = () => {
   198→    setWorkflowOpen(false);
   199→    onRefresh();
   200→  };
   201→
   202→  // Don't render if no record (drawer closed)
   203→  if (!record) return null;
   204→
   205→  return (
   206→    <Dialog open={open} onOpenChange={(o) => { if (!o) { setEditMode(false); onClose(); } }}>
   207→      <DialogContent className="max-w-[600px] w-full h-[90vh] max-h-[90vh] flex flex-col p-0 gap-0">
   208→        {/* Header */}
   209→        <div className="flex items-center gap-3 px-4 h-[60px] border-b border-[var(--erp-border)] bg-[var(--erp-bg-card)] shrink-0">
   210→          <div
   211→            className="w-9 h-9 rounded-lg flex items-center justify-center shrink-0"
   212→            style={{ background: `linear-gradient(135deg, ${register.color}30, ${register.color}10)`, color: register.color }}
   213→          >
   214→            <FAIcon name={register.icon} className="text-[14px]" />
   215→          </div>
   216→          <div className="flex-1 min-w-0">
   217→            <div className="text-[14px] font-semibold text-[var(--erp-text)] truncate" style={{ fontFamily: 'var(--font-display)' }}>
   218→              {register.name} — #{record.sequence}
   219→            </div>
   220→            <div className="text-[10px] text-[var(--erp-text-muted)] flex items-center gap-1.5">
   221→              <span>Record ID: {record.id.slice(-8)}</span>
   222→              {currentStatus && (
   223→                <>
   224→                  <span>·</span>
   225→                  <StatusPill status={currentStatus} />
   226→                </>
   227→              )}
   228→            </div>
   229→          </div>
   230→          {editMode && (
   231→            <div className="flex items-center gap-1 mr-2">
   232→              <span className="text-[10px] px-2 py-0.5 rounded-full bg-[var(--erp-accent-dim)] text-[var(--erp-accent)] font-semibold flex items-center gap-1">
   233→                <Pencil className="w-2.5 h-2.5" /> EDITING
   234→              </span>
   235→            </div>
   236→          )}
   237→          <button
   238→            type="button"
   239→            onClick={() => { setEditMode(false); onClose(); }}
   240→            className="p-2.5 rounded-lg bg-[var(--erp-bg-hover)] text-[var(--erp-text)] hover:bg-[var(--erp-danger)] hover:text-white transition-all shrink-0"
   241→            aria-label="Close"
   242→          >
   243→            <X className="w-4 h-4" />
   244→          </button>
   245→        </div>
   246→        {/* Action bar */}
   247→        <div className="flex items-center gap-1.5 px-4 py-2 border-b border-[var(--erp-border)] bg-[var(--erp-bg-card)] shrink-0">
   248→          {!editMode ? (
   249→            <>
   250→              <Button variant="outline" size="sm" onClick={handlePrint} className="h-8 text-[11px]">
   251→                <Printer className="w-3.5 h-3.5 mr-1" /> Print
   252→              </Button>
   253→              {hasStatusCol && (canApprove || canEdit) && (
   254→                <Button variant="outline" size="sm" onClick={handleWorkflowClick} className="h-8 text-[11px] border-[var(--erp-accent-border)] text-[var(--erp-accent)] hover:bg-[var(--erp-accent-dim)]">
   255→                  <WorkflowIcon className="w-3.5 h-3.5 mr-1" /> Workflow
   256→                </Button>
   257→              )}
   258→              {canEdit && (
   259→                <Button size="sm" onClick={enterEditMode} className="h-8 text-[11px] ml-auto bg-[var(--erp-accent)] hover:bg-[var(--erp-accent-hover)]">
   260→                  <Pencil className="w-3.5 h-3.5 mr-1" /> Inline Edit
   261→                </Button>
   262→              )}
   263→            </>
   264→          ) : (
   265→            <>
   266→              <Button variant="outline" size="sm" onClick={cancelEdit} disabled={saving} className="h-8 text-[11px]">
   267→                <XCircle className="w-3.5 h-3.5 mr-1" /> Cancel
   268→              </Button>
   269→              <Button size="sm" onClick={saveInlineEdit} disabled={saving} className="h-8 text-[11px] ml-auto bg-[var(--erp-success)] hover:bg-[var(--erp-success)]/90">
   270→                {saving ? <Loader2 className="w-3.5 h-3.5 mr-1 animate-spin" /> : <Check className="w-3.5 h-3.5 mr-1" />}
   271→                {saving ? 'Saving...' : 'Save Changes'}
   272→              </Button>
   273→            </>
   274→          )}
   275→        </div>
   276→
   277→        {/* Error banner (if validation errors) */}
   278→        {editMode && Object.keys(errors).length > 0 && (
   279→          <div className="px-4 py-2 border-b border-[var(--erp-danger)]/30 bg-[rgba(239,68,68,0.05)] flex items-center gap-2 text-[11px] text-[var(--erp-danger)]">
   280→            <AlertCircle className="w-3.5 h-3.5 shrink-0" />
   281→            <span>{Object.keys(errors).length} field(s) need attention</span>
   282→          </div>
   283→        )}
   284→
   285→        {/* Tabs (hidden in edit mode to maximize space) */}
   286→        {!editMode && (
   287→          <div className="flex items-center gap-1 px-4 py-1.5 border-b border-[var(--erp-border)] bg-[var(--erp-bg-card)] shrink-0">
   288→            <TabButton active={tab === 'details'} onClick={() => setTab('details')} icon={<FileText className="w-3.5 h-3.5" />} label="Details" />
   289→            <TabButton active={tab === 'history'} onClick={() => setTab('history')} icon={<HistoryIcon className="w-3.5 h-3.5" />} label="History" count={history.length} />
   290→            <TabButton active={tab === 'related'} onClick={() => setTab('related')} icon={<Link2 className="w-3.5 h-3.5" />} label="Related" />
   291→            <TabButton active={tab === 'activity'} onClick={() => setTab('activity')} icon={<Activity className="w-3.5 h-3.5" />} label="Activity" />
   292→          </div>
   293→        )}
   294→
   295→        {/* Content */}
   296→        <div className="flex-1 overflow-y-auto bg-[var(--erp-bg)]">
   297→          {editMode ? (
   298→            <InlineEditTab
   299→              register={register}
   300→              data={editData}
   301→              errors={errors}
   302→              masterData={masterData}
   303→              onChange={setField}
   304→              currency={currency}
   305→            />
   306→          ) : (
   307→            <>
   308→              {tab === 'details' && (
   309→                <DetailsTab register={register} record={record} currency={currency} onRefresh={onRefresh} />
   310→              )}
   311→              {tab === 'history' && (
   312→                <HistoryTab history={history} loading={historyLoading} hasStatusCol={!!hasStatusCol} />
   313→              )}
   314→              {tab === 'related' && (
   315→                <RelatedTab related={related} loading={relatedLoading} />
   316→              )}
   317→              {tab === 'activity' && (
   318→                <ActivityTab record={record} register={register} />
   319→              )}
   320→            </>
   321→          )}
   322→        </div>
   323→
   324→        {/* Footer with CLOSE button */}
   325→        <div className="px-4 py-2 border-t border-[var(--erp-border)] bg-[var(--erp-bg-card)] shrink-0 flex items-center justify-between text-[10px] text-[var(--erp-text-muted)]">
   326→          <span className="flex items-center gap-1">
   327→            <Clock className="w-3 h-3" />
   328→            Created {formatDate(record.createdAt)}
   329→          </span>
   330→          <span>Updated {formatTimeAgo(record.updatedAt)}</span>
   331→        </div>
   332→        {/* Big visible CLOSE bar — always works */}
   333→        <button
   334→          type="button"
   335→          onClick={() => { setEditMode(false); onClose(); }}
   336→          className="w-full py-3 bg-[var(--erp-accent)] text-white text-[12px] font-semibold hover:bg-[var(--erp-accent-hover)] transition-colors shrink-0"
   337→        >
   338→          ✕ Close
   339→        </button>
   340→
   341→        {/* Workflow modal */}
   342→        <ApprovalWorkflow
   343→          open={workflowOpen}
   344→          register={register}
   345→          record={record}
   346→          onClose={() => setWorkflowOpen(false)}
   347→          onTransition={handleWorkflowTransition}
   348→        />
   349→      </DialogContent>
   350→    </Dialog>
   351→  );
   352→}

   354→// ---------- Inline Edit Tab ----------
   355→function InlineEditTab({
   356→  register, data, errors, masterData, onChange, currency = 'AED',
   357→}: {
   358→  register: Register;
   359→  data: Record<string, any>;
   360→  errors: Record<string, string>;
   361→  masterData: Record<string, string[]>;
   362→  onChange: (name: string, value: any) => void;
   363→  currency?: string;
   364→}) {
   365→  const cols = register.columns.filter((c) => c.type !== 'auto_increment');
   366→
   367→  // Group columns by type for better layout (same as DetailsTab)
   368→  const mainFields = cols.filter((c) => ['text', 'date', 'datetime', 'time', 'number', 'currency', 'percentage', 'email', 'phone', 'dropdown', 'status', 'priority', 'rating', 'employee', 'department', 'building', 'asset', 'equipment', 'vendor', 'url', 'color'].includes(c.type));
   369→  const longFields = cols.filter((c) => c.type === 'long_text');
   370→  const multiFields = cols.filter((c) => c.type === 'multi_select' || c.type === 'tags');
   371→  const imageFields = cols.filter((c) => c.type === 'image');
   372→
   373→  return (
   374→    <div className="p-4 space-y-5">
   375→      <div className="flex items-center gap-2 p-2.5 rounded-md bg-[var(--erp-accent-dim)] border border-[var(--erp-accent-border)] text-[11px] text-[var(--erp-accent)]">
   376→        <Pencil className="w-3.5 h-3.5 shrink-0" />
   377→        <span>Edit fields directly. Changes are saved when you click "Save Changes".</span>
   378→      </div>
   379→
   380→      {/* Image gallery (inline-edit) */}
   381→      {imageFields.length > 0 && (
   382→        <div>
   383→          <SectionLabel icon="fa-image" label="Images" count={imageFields.length} />
   384→          <div className="grid grid-cols-2 sm:grid-cols-3 gap-2.5">
   385→            {imageFields.map((col) => (
   386→              <InlineField key={col.name} col={col} value={data[col.name]} error={errors[col.name]} masterData={masterData} onChange={(v) => onChange(col.name, v)} currency={currency} />
   387→            ))}
   388→          </div>
   389→        </div>
   390→      )}
   391→
   392→      {/* Main fields grid */}
   393→      <div>
   394→        <SectionLabel icon="fa-circle-info" label="Record Fields" count={mainFields.length} />
   395→        <div className="grid grid-cols-2 gap-2.5">
   396→          {mainFields.map((col) => (
   397→            <InlineField key={col.name} col={col} value={data[col.name]} error={errors[col.name]} masterData={masterData} onChange={(v) => onChange(col.name, v)} currency={currency} />
   398→          ))}
   399→        </div>
   400→      </div>
   401→
   402→      {/* Multi-select fields */}
   403→      {multiFields.length > 0 && (
   404→        <div>
   405→          <SectionLabel icon="fa-list-check" label="Tags & Categories" count={multiFields.length} />
   406→          <div className="space-y-2.5">
   407→            {multiFields.map((col) => (
   408→              <InlineField key={col.name} col={col} value={data[col.name]} error={errors[col.name]} masterData={masterData} onChange={(v) => onChange(col.name, v)} fullWidth currency={currency} />
   409→            ))}
   410→          </div>
   411→        </div>
   412→      )}
   413→
   414→      {/* Long text fields */}
   415→      {longFields.length > 0 && (
   416→        <div>
   417→          <SectionLabel icon="fa-align-left" label="Notes & Descriptions" count={longFields.length} />
   418→          <div className="space-y-2.5">
   419→            {longFields.map((col) => (
   420→              <InlineField key={col.name} col={col} value={data[col.name]} error={errors[col.name]} masterData={masterData} onChange={(v) => onChange(col.name, v)} fullWidth currency={currency} />
   421→            ))}
   422→          </div>
   423→        </div>
   424→      )}
   425→    </div>
   426→  );
   427→}
   428→
   429→function InlineField({
   430→  col, value, error, masterData, onChange, fullWidth, currency = 'AED',
   431→}: {
   432→  col: ColumnDef;
   433→  value: any;
   434→  error?: string;
   435→  masterData: Record<string, string[]>;
   436→  onChange: (v: any) => void;
   437→  fullWidth?: boolean;
   438→  currency?: string;
   439→}) {
   440→  const label = (
   441→    <div className="text-[9px] uppercase tracking-wide text-[var(--erp-text-muted)] mb-1 flex items-center gap-1">
   442→      <FAIcon name={colIconFor(col.type)} className="text-[8px]" />
   443→      <span>{displayColumnName(col.name, currency)}</span>
   444→      {col.required && <span className="text-[var(--erp-danger)]">*</span>}
   445→    </div>
   446→  );
   447→
   448→  const errorEl = error ? (
   449→    <p className="text-[10px] text-[var(--erp-danger)] mt-0.5 flex items-center gap-1">
   450→      <AlertCircle className="w-3 h-3" /> {error}
   451→    </p>
   452→  ) : null;
   453→
   454→  const wrapperClass = cn(
   455→    'border border-[var(--erp-border)] rounded-md p-2.5 bg-[var(--erp-bg-card)] focus-within:border-[var(--erp-accent-border)] focus-within:ring-1 focus-within:ring-[var(--erp-accent-border)] transition-all',
   456→    fullWidth && 'col-span-2',
   457→    error && 'border-[var(--erp-danger)]/40',
   458→  );
   459→
   460→  switch (col.type) {
   461→    case 'long_text':
   462→      return (
   463→        <div className={wrapperClass}>
   464→          {label}
   465→          <Textarea
   466→            value={value || ''}
   467→            onChange={(e) => onChange(e.target.value)}
   468→            rows={3}
   469→            className="text-[12px] resize-y bg-[var(--erp-bg-input)] border-0 p-0 focus-visible:ring-0"
   470→            placeholder={`Enter ${col.name.toLowerCase()}...`}
   471→          />
   472→          {errorEl}
   473→        </div>
   474→      );
   475→    case 'dropdown':
   476→    case 'status':
   477→    case 'priority':
   478→      return (
   479→        <div className={wrapperClass}>
   480→          {label}
   481→          <Select value={value || ''} onValueChange={onChange}>
   482→            <SelectTrigger className="text-[12px] h-8 bg-[var(--erp-bg-input)] border-0 p-0 focus:ring-0">
   483→              <SelectValue placeholder={`Select...`} />
   484→            </SelectTrigger>
   485→            <SelectContent>
   486→              {(col.options || []).map((o) => (
   487→                <SelectItem key={o} value={o} className="text-[12px]">{o}</SelectItem>
   488→              ))}
   489→            </SelectContent>
   490→          </Select>
   491→          {errorEl}
   492→        </div>
   493→      );
   494→    case 'employee':
   495→    case 'department':
   496→    case 'building':
   497→    case 'asset':

   721→function MultiSelectInline({ options, value, onChange }: { options: string[]; value: string[]; onChange: (v: string[]) => void }) {
   722→  const toggle = (opt: string) => {
   723→    if (value.includes(opt)) onChange(value.filter((v) => v !== opt));
   724→    else onChange([...value, opt]);
   725→  };
   726→  return (
   727→    <div className="flex flex-wrap gap-1.5 min-h-[32px]">
   728→      {options.length === 0 ? (
   729→        <span className="text-[11px] text-[var(--erp-text-muted)]">No options available</span>
   730→      ) : (
   731→        options.map((opt) => {
   732→          const selected = value.includes(opt);
   733→          return (
   734→            <button
   735→              key={opt}
   736→              type="button"
   737→              onClick={() => toggle(opt)}
   738→              className={cn(
   739→                'text-[11px] px-2 py-1 rounded-md transition-colors',
   740→                selected
   741→                  ? 'bg-[var(--erp-accent)] text-white'
   742→                  : 'bg-[var(--erp-bg-card)] text-[var(--erp-text-secondary)] hover:bg-[var(--erp-bg-hover)]',
   743→              )}
   744→            >
   745→              {selected && '✓ '}
   746→              {opt}
   747→            </button>
   748→          );
   749→        })
   750→      )}
   751→    </div>
   752→  );
   753→}

   755→// Image upload field for the drawer's inline edit mode
   756→function DrawerImageField({ value, onChange }: { value: any; onChange: (v: any) => void }) {
   757→  const [uploading, setUploading] = useState(false);
   758→  const fileInputRef = useRef<HTMLInputElement>(null);
   759→
   760→  const handleFile = async (file: File) => {
   761→    if (!file) return;
   762→    setUploading(true);
   763→    try {
   764→      const result = await uploadsApi.upload(file);
   765→      onChange(result.url);
   766→      toast.success('Image uploaded');
   767→    } catch (e: any) {
   768→      toast.error('Upload failed', { description: e.message });
   769→    } finally {
   770→      setUploading(false);
   771→    }
   772→  };
   773→
   774→  return (
   775→    <div className="flex items-center gap-2">
   776→      <input
   777→        ref={fileInputRef}
   778→        type="file"
   779→        accept="image/*"
   780→        className="hidden"
   781→        onChange={(e) => { const f = e.target.files?.[0]; if (f) handleFile(f); e.target.value = ''; }}
   782→      />
   783→      {value ? (
   784→        <div className="relative">
   785→          <img src={String(value)} alt="preview" className="w-16 h-16 object-cover rounded-md border border-[var(--erp-border)]" />

   956→// ---------- History Tab ----------
   957→function HistoryTab({ history, loading, hasStatusCol }: { history: HistoryEntry[]; loading: boolean; hasStatusCol: boolean }) {
   958→  const exportHistoryCsv = () => {
   959→    if (history.length === 0) return;
   960→    const headers = ['Date', 'User', 'Action', 'From Status', 'To Status', 'Summary'];
   961→    const rows = history.map((entry) => [
   962→      new Date(entry.createdAt).toLocaleString(),
   963→      entry.userName,
   964→      entry.action,
   965→      entry.statusChange?.from || '',
   966→      entry.statusChange?.to || '',
   967→      entry.summary.replace(/"/g, '""'),
   968→    ].map((v) => `"${v}"`));
   969→    const csv = [headers.map((h) => `"${h}"`).join(','), ...rows.map((r) => r.join(','))].join('\n');
   970→    const blob = new Blob([csv], { type: 'text/csv' });
   971→    const url = URL.createObjectURL(blob);
   972→    const a = document.createElement('a');
   973→    a.href = url;
   974→    a.download = `history_${new Date().toISOString().slice(0, 10)}.csv`;
   975→    a.click();
   976→    URL.revokeObjectURL(url);
   977→    toast.success(`Exported ${history.length} history entries`);
   978→  };
   979→
   980→  if (loading) {
   981→    return (
   982→      <div className="flex items-center justify-center py-12">
   983→        <Loader2 className="w-6 h-6 text-[var(--erp-accent)] animate-spin" />
   984→      </div>
   985→    );
   986→  }
   987→
   988→  if (history.length === 0) {
   989→    return (
   990→      <div className="flex flex-col items-center justify-center py-12 px-4 text-center">

  1300→function formatCurrencyCompact(n: number, currency = 'AED'): string {
