  1220→function initRegisters() {
  1221→  REGISTER_DEFS.forEach(def => {
  1222→    const reg = { ...def, records: def.records.map((r, i) => {
  1223→      // Auto-generate auto_increment fields
  1224→      const rec = { ...r };
  1225→      def.columns.forEach(col => {
  1226→        if (col.type === 'auto_increment' && !rec[col.name]) rec[col.name] = i + 1;
  1227→      });
  1228→      return rec;
  1229→    })};
  1230→    state.registers[def.id] = reg;
  1231→  });
  1232→}

  1337→function activateTab(id) {

  1341→  state.filters = {};

  1345→  highlightSidebarItem(tab.type === 'dashboard' ? '__dashboard' : tab.regId);
  1346→  document.getElementById('tabActions').style.display = tab.type === 'register' ? '' : 'none';

  1348→    renderDashboard();

  1355→    const reg = state.registers[tab.regId];
  1356→    setBreadcrumb(`Registers > ${reg.name}`);

  1473→function renderCell(value, col) {
  1474→  if (value == null || value === '') return '<span style="color:var(--text-muted)">—</span>';
  1475→  switch (col.type) {
  1476→    case 'status': {
  1477→      const c = STATUS_COLORS[value] || 'gray';

  1480→    case 'priority': {
  1481→      const cls = PRIORITY_CLASSES[value] || 'low';
  1482→      return `<span class="priority-dot ${cls}">${value}</span>`;

  1485→      return `<input type="checkbox" ${value?'checked':''} disabled style="accent-color:var(--accent)">`;
  1486→    case 'currency':
  1487→      return `<span class="currency-cell">${Number(value).toLocaleString('en-AE', {style:'currency',currency:'AED',minimumFractionDigits:0})}</span>`;

  1497→    }
  1498→    case 'date':
  1499→      return value ? new Date(value).toLocaleDateString('en-GB', {day:'2-digit',month:'short',year:'numeric'}) : '';
  1500→    case 'time':
  1501→      return value || '';
  1502→    case 'datetime':
  1503→      return value ? new Date(value).toLocaleString('en-GB', {day:'2-digit',month:'short',year:'numeric',hour:'2-digit',minute:'2-digit'}) : '';

  1518→  }

  1521→function getProcessedRecords(regId) {
  1522→  const reg = state.registers[regId];
  1523→  let records = [...reg.records];
  1524→  // Apply filters
  1525→  const q = document.getElementById('quickSearch_'+regId);
  1526→  const quickQ = q ? q.value.toLowerCase() : '';
  1527→  Object.entries(state.filters).forEach(([colName, val]) => {
  1528→    if (!val) return;
  1529→    const lv = val.toLowerCase();
  1530→    records = records.filter(r => {
  1531→      const rv = r[colName];
  1532→      if (Array.isArray(rv)) return rv.some(v => String(v).toLowerCase().includes(lv));
  1533→      return String(rv ?? '').toLowerCase().includes(lv);
  1534→    });
  1535→  });
  1536→  if (quickQ) {
  1537→    records = records.filter(r => Object.values(r).some(v => {
  1538→      if (Array.isArray(v)) return v.some(x => String(x).toLowerCase().includes(quickQ));
  1539→      return String(v ?? '').toLowerCase().includes(quickQ);
  1540→    }));
  1541→  }
  1542→  // Apply sort
  1543→  if (state.sortCol >= 0) {
  1544→    const col = reg.columns[state.sortCol];
  1545→    records.sort((a, b) => {
  1546→      let va = a[col.name], vb = b[col.name];
  1547→      if (va == null) va = '';
  1548→      if (vb == null) vb = '';
  1549→      if (col.type === 'number' || col.type === 'currency' || col.type === 'percentage' || col.type === 'auto_increment') {
  1550→        return (Number(va) || 0) - (Number(vb) || 0);
  1551→      }
  1552→      if (col.type === 'date' || col.type === 'datetime') {
  1553→        return new Date(va || 0) - new Date(vb || 0);
  1554→      }
  1555→      return String(va).localeCompare(String(vb));
  1556→    });
  1557→    if (state.sortDir === 'desc') records.reverse();
  1558→  }
  1559→  return records;
  1560→}
