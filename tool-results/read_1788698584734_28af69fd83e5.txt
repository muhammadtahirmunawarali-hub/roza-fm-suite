    34→
    35→interface Props {
    36→  registerId: string;
    37→}
    38→
    39→const PAGE_SIZES = [10, 25, 50, 100];
    40→
    41→export function RegisterView({ registerId }: Props) {
    42→  const { hasPermission, user, currency } = useErpStore();
    43→  const [register, setRegister] = useState<Register | null>(null);
    44→  const [records, setRecords] = useState<RecordData[]>([]);
    45→  const [total, setTotal] = useState(0);
    46→  const [page, setPage] = useState(1);
    47→  const [pageSize, setPageSize] = useState(25);
    48→  const [search, setSearch] = useState('');
    49→  const [debouncedSearch, setDebouncedSearch] = useState('');
    50→  const [sortField, setSortField] = useState<string>('');
    51→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc');
    52→  const [filters, setFilters] = useState<Record<string, string>>({});
    53→  const [showFilters, setShowFilters] = useState(false);
    54→  const [loading, setLoading] = useState(true);
    55→  const [formOpen, setFormOpen] = useState(false);
    56→  const [importOpen, setImportOpen] = useState(false);
    57→  const [editing, setEditing] = useState<RecordData | null>(null);
    58→  const [viewing, setViewing] = useState<RecordData | null>(null);
    59→  const [workflowTarget, setWorkflowTarget] = useState<RecordData | null>(null);
    60→  const [deleteTarget, setDeleteTarget] = useState<RecordData | null>(null);
    61→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
    62→  const [company, setCompany] = useState({ name: 'FMCore Facilities Management', address: '', phone: '', email: '', tax_number: '' });
    63→  const [hiddenColumns, setHiddenColumns] = useState<Set<string>>(new Set());
    64→  const [showColumnToggle, setShowColumnToggle] = useState(false);
    65→  const [columnEditorOpen, setColumnEditorOpen] = useState(false);
    66→
    67→  // Permission flags (register may be null initially)
    68→  const regCode = register?.code || '';
    69→  const canView = hasPermission(regCode || registerId, 'view');
    70→  const canCreate = hasPermission(regCode, 'create');
    71→  const canEdit = hasPermission(regCode, 'edit');
    72→  const canDelete = hasPermission(regCode, 'delete');
    73→  const canExport = hasPermission(regCode, 'export');
    74→  const canImport = hasPermission(regCode, 'import');
    75→  const canApprove = hasPermission(regCode, 'approve');
    76→  const hasStatusCol = register?.columns.some((c) => c.type === 'status');
    77→
    78→  // Visible columns (filtered by user's column visibility preferences)
    79→  const visibleColumns = useMemo(() => {
    80→    if (!register) return [];
    81→    return register.columns.filter((c) => !hiddenColumns.has(c.name));
    82→  }, [register, hiddenColumns]);
    83→
    84→  const toggleColumn = (colName: string) => {
    85→    setHiddenColumns((prev) => {
    86→      const next = new Set(prev);
    87→      if (next.has(colName)) next.delete(colName);
    88→      else next.add(colName);
    89→      return next;
    90→    });
    91→  };
    92→
    93→  // Load register meta + company settings
    94→  useEffect(() => {
    95→    let cancelled = false;
    96→    (async () => {
    97→      try {
    98→        const [regs, settings] = await Promise.all([registersApi.list(), fetch('/api/erp/settings').then((r) => r.json())]);
    99→        const r = regs.find((x) => x.id === registerId);
   100→        if (!cancelled && r) {
   101→          setRegister(r);
   102→          const autoCol = r.columns.find((c) => c.type === 'auto_increment');
   103→          if (autoCol) { setSortField(autoCol.name); setSortDir('asc'); }
   104→        }
   105→        if (!cancelled && Array.isArray(settings)) {
   106→          const map: Record<string, string> = {};
   107→          settings.forEach((s: any) => (map[s.key] = s.value));
   108→          setCompany({
   109→            name: map['company.name'] || 'FMCore Facilities Management',
   110→            address: map['company.address'] || '',
   111→            phone: map['company.phone'] || '',
   112→            email: map['company.email'] || '',
   113→            tax_number: map['company.tax_number'] || '',
   114→          });
   115→        }
   116→      } catch (e) { console.error(e); }
   117→    })();
   118→    return () => { cancelled = true; };
   119→  }, [registerId]);
   120→
   121→  // Debounce search
   122→  useEffect(() => {
   123→    const t = setTimeout(() => { setDebouncedSearch(search); setPage(1); }, 350);
   124→    return () => clearTimeout(t);
   125→  }, [search]);
   126→
   127→  // Clear selection when page/filters change
   128→  useEffect(() => { setSelectedIds(new Set()); }, [page, debouncedSearch, filters, sortField, sortDir]);
   129→
   130→  const loadRecords = useCallback(async () => {
   131→    if (!register) return;
   132→    setLoading(true);
   133→    try {
   134→      const res = await recordsApi.list(registerId, {
   135→        page, pageSize, search: debouncedSearch, sortField, sortDir, filters,
   136→      });
   137→      setRecords(res.data);
   138→      setTotal(res.total);
   139→    } catch (e: any) {
   140→      toast.error('Failed to load records', { description: e.message });
   141→    } finally {
   142→      setLoading(false);
   143→    }
   144→  }, [register, registerId, page, pageSize, debouncedSearch, sortField, sortDir, filters]);
   145→
   146→  useEffect(() => { loadRecords(); }, [loadRecords]);
   147→
   148→  const toggleSort = (field: string) => {
   149→    if (sortField === field) {
   150→      setSortDir((d) => (d === 'asc' ? 'desc' : 'asc'));
   151→    } else {
   152→      setSortField(field);
   153→      setSortDir('asc');
   154→    }
   155→  };
   156→
   157→  const handleSaved = () => {
   158→    setFormOpen(false);
   159→    setEditing(null);
   160→    loadRecords();
   161→  };
   162→
   163→  const handleDelete = async () => {
   164→    if (!deleteTarget) return;
   165→    try {
   166→      await recordsApi.remove(registerId, deleteTarget.id);
   167→      toast.success('Record deleted');
   168→      setDeleteTarget(null);
   169→      loadRecords();
   170→    } catch (e: any) {
   171→      toast.error('Delete failed', { description: e.message });
   172→    }
   173→  };
   174→
   175→  const exportCsv = () => {
   176→    if (!register || records.length === 0) return;
   177→    const headers = register.columns.map((c) => c.name);
   178→    const rows = records.map((r) =>
   179→      register.columns.map((c) => {
   180→        const v = r.data[c.name];
   181→        if (Array.isArray(v)) return `"${v.join(', ')}"`;
   182→        if (v === undefined || v === null) return '';
   183→        return `"${String(v).replace(/"/g, '""')}"`;
   184→      }),
   185→    );
   186→    const csv = [headers.join(','), ...rows.map((r) => r.join(','))].join('\n');
   187→    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8' });
   188→    const url = URL.createObjectURL(blob);
   189→    const a = document.createElement('a');
   190→    a.href = url;
   191→    a.download = `${register.code}_${new Date().toISOString().slice(0, 10)}.csv`;
   192→    a.click();
   193→    URL.revokeObjectURL(url);
   194→    toast.success(`Exported ${records.length} records to CSV`);
   195→  };
   196→
   197→  const exportJson = () => {
   198→    if (!register || records.length === 0) return;
   199→    const exportData = {
   200→      register: {
   201→        name: register.name,
   202→        code: register.code,
   203→        category: register.category,
   204→        columns: register.columns,
   205→      },
   206→      exportedAt: new Date().toISOString(),
   207→      recordCount: records.length,
   208→      records: records.map((r) => ({
   209→        id: r.id,
   210→        sequence: r.sequence,
   211→        data: r.data,
   212→        createdAt: r.createdAt,
   213→        updatedAt: r.updatedAt,
   214→        createdBy: r.createdBy,
   215→        updatedBy: r.updatedBy,
   216→      })),
   217→    };
   218→    const blob = new Blob([JSON.stringify(exportData, null, 2)], { type: 'application/json' });
   219→    const url = URL.createObjectURL(blob);
   220→    const a = document.createElement('a');
   221→    a.href = url;
   222→    a.download = `${register.code}_${new Date().toISOString().slice(0, 10)}.json`;
   223→    a.click();
   224→    URL.revokeObjectURL(url);
   225→    toast.success(`Exported ${records.length} records to JSON`);
   226→  };
   227→
   228→  const handlePrint = (recs: RecordData[]) => {
   229→    if (!register) return;
   230→    if (recs.length === 1) {
   231→      printRecord(register, recs[0], company);
   232→    } else {
   233→      // For multiple, print first one (limitation noted in UI)
   234→      recs.slice(0, 5).forEach((r, i) => {
   235→        setTimeout(() => printRecord(register, r, company), i * 400);
   236→      });
   237→    }
   238→  };
   239→
   240→  const totalPages = Math.max(1, Math.ceil(total / pageSize));
   241→
   242→  // Selection helpers
   243→  const toggleSelect = (id: string) => {
   244→    setSelectedIds((s) => {
   245→      const next = new Set(s);
   246→      if (next.has(id)) next.delete(id); else next.add(id);
   247→      return next;
   248→    });
   249→  };
   250→  const toggleSelectAll = () => {
   251→    if (selectedIds.size === records.length) {
   252→      setSelectedIds(new Set());
   253→    } else {
   254→      setSelectedIds(new Set(records.map((r) => r.id)));
   255→    }
   256→  };
   257→  const selectedRecords = records.filter((r) => selectedIds.has(r.id));
   258→
   259→  // Compute quick stats per register
   260→  const stats = useMemo(() => {
   261→    if (!register) return null;
   262→    const statusCol = register.columns.find((c) => c.type === 'status');
   263→    const priorityCol = register.columns.find((c) => c.type === 'priority');
   264→    const currencyCol = register.columns.find((c) => c.type === 'currency');
   265→    const out: { label: string; value: string | number; color: string }[] = [];
   266→    out.push({ label: 'Total', value: total, color: 'var(--erp-text-secondary)' });
   267→    if (statusCol && statusCol.options) {
   268→      statusCol.options.slice(0, 3).forEach((opt) => {
   269→        const count = records.filter((r) => r.data[statusCol.name] === opt).length;
   270→        out.push({ label: opt, value: count, color: statusColorFor(opt) });
   271→      });
   272→    }
   273→    if (priorityCol && priorityCol.options) {
   274→      const critical = records.filter((r) => r.data[priorityCol.name] === 'Critical').length;
   275→      if (critical > 0) out.push({ label: 'Critical', value: critical, color: 'var(--erp-danger)' });
   276→    }
   277→    if (currencyCol) {
   278→      const sum = records.reduce((s, r) => s + (Number(r.data[currencyCol.name]) || 0), 0);
   279→      out.push({ label: currencyCol.name + ' (Σ)', value: formatCurrencyCompact(sum, currency), color: 'var(--erp-accent)' });
   280→    }
   281→    return out.slice(0, 6);
   282→  }, [register, records, total]);
   283→
   284→  if (!register) {
   285→    return <div className="p-6 text-[var(--erp-text-muted)]">Loading register...</div>;
   286→  }
   287→
   288→  const filterableCols = register.columns.filter((c) =>
   289→    ['status', 'priority', 'dropdown'].includes(c.type) && c.options && c.options.length > 0
   290→  );
   291→
   292→  return (
   293→    <div className="flex flex-col h-full">
   294→      {/* Header */}
   295→      <div className="px-4 md:px-6 py-3 border-b border-[var(--erp-border)] bg-[var(--erp-bg-card)]">
   296→        <div className="flex items-start justify-between flex-wrap gap-3">
   297→          <div className="flex items-start gap-3">
   298→            <div
   299→              className="w-10 h-10 rounded-lg flex items-center justify-center shrink-0"
   300→              style={{ background: `linear-gradient(135deg, ${register.color}30, ${register.color}10)`, color: register.color }}
   301→            >
   302→              <FAIcon name={register.icon} className="text-[16px]" />
   303→            </div>
   304→            <div>
   305→              <h1 className="text-[18px] font-semibold text-[var(--erp-text)]" style={{ fontFamily: 'var(--font-display)' }}>
   306→                {register.name}
   307→              </h1>
   308→              {register.description && (
   309→                <p className="text-[11px] text-[var(--erp-text-muted)] mt-0.5">{register.description}</p>
   310→              )}
   311→              <div className="flex items-center gap-2 mt-1 text-[10px] text-[var(--erp-text-muted)]">
   312→                <span className="px-1.5 py-0.5 rounded font-medium" style={{ background: register.color + '20', color: register.color }}>{register.category}</span>
   313→                <span>·</span>
   314→                <span>{register.columns.length} columns</span>
   315→                <span>·</span>
   316→                <span>{total} records</span>
   317→              </div>
   318→            </div>
   319→          </div>
   320→
   321→          <div className="flex items-center gap-1.5 flex-wrap">
   322→            <SavedViews
   323→              registerId={register.id}
   324→              registerName={register.name}
   325→              currentFilters={{ search: debouncedSearch, filters, sortField, sortDir }}
   326→              onApply={(v) => {
   327→                setSearch(v.search || '');
   328→                setFilters(v.filters || {});
   329→                setSortField(v.sortField || '');
   330→                setSortDir(v.sortDir || 'asc');
   331→                setPage(1);
   332→              }}
   333→            />
   334→            <Button variant="outline" size="sm" onClick={() => setShowFilters((v) => !v)} className="h-8 text-[12px]">
   335→              <Filter className="w-3.5 h-3.5 mr-1" /> Filters
   336→              {Object.values(filters).filter(Boolean).length > 0 && (
   337→                <span className="ml-1 px-1.5 rounded-full bg-[var(--erp-accent)] text-white text-[9px]">
   338→                  {Object.values(filters).filter(Boolean).length}
   339→                </span>
   340→              )}
   341→            </Button>
   342→            {/* Column Visibility Toggle */}
   343→            <div className="relative">
   344→              <Button variant="outline" size="sm" onClick={() => setShowColumnToggle((v) => !v)} className="h-8 text-[12px]">
   345→                <Columns3 className="w-3.5 h-3.5 mr-1" /> Columns
   346→                {hiddenColumns.size > 0 && (
   347→                  <span className="ml-1 px-1.5 rounded-full bg-[var(--erp-accent)] text-white text-[9px]">
   348→                    {hiddenColumns.size}
   349→                  </span>
   350→                )}
   351→              </Button>
   352→              {showColumnToggle && (
   353→                <div className="absolute top-full left-0 mt-1 w-56 max-h-[300px] overflow-y-auto bg-[var(--erp-bg-card)] border border-[var(--erp-border)] rounded-md shadow-xl z-30 p-1.5">
   354→                  <div className="text-[10px] uppercase tracking-wide text-[var(--erp-text-muted)] px-2 py-1 mb-1 border-b border-[var(--erp-border)]">Toggle Columns</div>
   355→                  {register.columns.map((col) => (
   356→                    <label key={col.name} className="flex items-center gap-2 px-2 py-1.5 rounded hover:bg-[var(--erp-bg-hover)] cursor-pointer text-[11px]">
   357→                      <Checkbox
   358→                        checked={!hiddenColumns.has(col.name)}
   359→                        onCheckedChange={() => toggleColumn(col.name)}
   360→                      />
   361→                      <span className="text-[var(--erp-text-secondary)] flex-1 truncate">{col.name}</span>
   362→                      <span className="text-[9px] text-[var(--erp-text-muted)]">{col.type.replace('_', ' ')}</span>
   363→                    </label>
   364→                  ))}
   365→                  {hiddenColumns.size > 0 && (
   366→                    <button
   367→                      onClick={() => setHiddenColumns(new Set())}
   368→                      className="w-full text-[10px] text-[var(--erp-accent)] hover:underline py-1.5 mt-1 border-t border-[var(--erp-border)]"
   369→                    >
   370→                      Show all columns
   371→                    </button>
   372→                  )}
   373→                </div>
   374→              )}
   375→            </div>
   376→            {/* Edit Columns (rename/retype/add/delete) */}
   377→            {canEdit && (
   378→              <Button variant="outline" size="sm" onClick={() => setColumnEditorOpen(true)} className="h-8 text-[12px]" title="Edit column structure">
   379→                <Settings2 className="w-3.5 h-3.5 mr-1" /> Edit
   380→              </Button>
   381→            )}
   382→            {canImport && (
   383→              <Button variant="outline" size="sm" onClick={() => setImportOpen(true)} className="h-8 text-[12px]">
   384→                <Upload className="w-3.5 h-3.5 mr-1" /> Import
   385→              </Button>
   386→            )}
   387→            {canExport && (
   388→              <div className="relative group">
   389→                <Button variant="outline" size="sm" className="h-8 text-[12px]">
   390→                  <Download className="w-3.5 h-3.5 mr-1" /> Export
   391→                  <ChevronDown className="w-3 h-3 ml-0.5" />
   392→                </Button>
   393→                <div className="absolute top-full right-0 mt-1 w-40 bg-[var(--erp-bg-card)] border border-[var(--erp-border)] rounded-md shadow-lg z-30 opacity-0 invisible group-hover:opacity-100 group-hover:visible transition-all">
   394→                  <button onClick={exportCsv} className="w-full flex items-center gap-2 px-3 py-2 text-[11px] text-[var(--erp-text-secondary)] hover:bg-[var(--erp-bg-hover)] hover:text-[var(--erp-text)] transition-colors border-b border-[var(--erp-border)]">
   395→                    <FileText className="w-3 h-3" /> Export as CSV
   396→                  </button>
   397→                  <button onClick={exportJson} className="w-full flex items-center gap-2 px-3 py-2 text-[11px] text-[var(--erp-text-secondary)] hover:bg-[var(--erp-bg-hover)] hover:text-[var(--erp-text)] transition-colors">
   398→                    <Braces className="w-3 h-3" /> Export as JSON
   399→                  </button>
   400→                </div>
   401→              </div>
   402→            )}
   403→            {canCreate && (
   404→              <Button
   405→                size="sm"
   406→                onClick={() => { setEditing(null); setFormOpen(true); }}
   407→                className="h-8 text-[12px] bg-[var(--erp-accent)] hover:bg-[var(--erp-accent-hover)]"
   408→              >
   409→                <Plus className="w-3.5 h-3.5 mr-1" /> Add Record
   410→              </Button>
   411→            )}
   412→          </div>
   413→        </div>
   414→
   415→        {/* Permission notice for read-only users */}
   416→        {!canCreate && !canEdit && !canDelete && canView && (
   417→          <div className="mt-2 flex items-center gap-1.5 text-[10px] text-[var(--erp-text-muted)]">
   418→            <FAIcon name="fa-eye" className="text-[9px]" />
   419→            <span>Read-only access — you can view records but not modify them</span>
   420→          </div>
   421→        )}
   422→
   423→        {/* Stats strip */}
   424→        {stats && stats.length > 0 && (
   425→          <div className="flex items-center gap-4 mt-3 text-[11px] flex-wrap">
   426→            {stats.map((s, i) => (
   427→              <div key={i} className="flex items-center gap-1.5">
   428→                <span className="w-2 h-2 rounded-full" style={{ background: s.color }} />
   429→                <span className="text-[var(--erp-text-muted)]">{s.label}:</span>
   430→                <span className="font-semibold text-[var(--erp-text)]">{s.value}</span>
   431→              </div>
   432→            ))}
   433→          </div>
   434→        )}
   435→
   436→        {/* Quick Status Filter Pills */}
   437→        {hasStatusCol && filterableCols.length > 0 && (() => {
   438→          const statusCol = register.columns.find((c) => c.type === 'status');
   439→          if (!statusCol?.options) return null;
   440→          const activeFilter = filters[statusCol.name];
   441→          return (
   442→            <div className="flex items-center gap-1.5 mt-2 flex-wrap">
   443→              <span className="text-[10px] uppercase tracking-wide text-[var(--erp-text-muted)] mr-1">Quick Filter:</span>
   444→              <button
   445→                onClick={() => { setFilters((f) => { const n = { ...f }; delete n[statusCol.name]; return n; }); setPage(1); }}
   446→                className={cn(
   447→                  'px-2 py-0.5 rounded-full text-[10px] font-medium transition-colors',
   448→                  !activeFilter
   449→                    ? 'bg-[var(--erp-accent)] text-white'
   450→                    : 'bg-[var(--erp-bg-input)] text-[var(--erp-text-secondary)] hover:bg-[var(--erp-bg-hover)]',
   451→                )}
   452→              >
   453→                All
   454→              </button>
   455→              {statusCol.options.map((opt) => {
   456→                const count = records.filter((r) => r.data[statusCol.name] === opt).length;
   457→                if (count === 0) return null;
   458→                return (
   459→                  <button
   460→                    key={opt}
   461→                    onClick={() => { setFilters((f) => ({ ...f, [statusCol.name]: activeFilter === opt ? '' : opt })); setPage(1); }}
   462→                    className={cn(
   463→                      'px-2 py-0.5 rounded-full text-[10px] font-medium transition-colors flex items-center gap-1',
   464→                      activeFilter === opt
   465→                        ? 'bg-[var(--erp-accent)] text-white'
   466→                        : 'bg-[var(--erp-bg-input)] text-[var(--erp-text-secondary)] hover:bg-[var(--erp-bg-hover)]',

   784→function formatCurrencyCompact(n: number, currency = 'AED'): string {

   790→function CellContent({ value, col, sequence, registerCode, expanded, currency = 'AED' }: { value: any; col: ColumnDef; sequence: number; registerCode: string; expanded?: boolean; currency?: string }) {
   791→  if (col.type === 'auto_increment') {
   792→    const prefix = guessPrefix(col.name, registerCode);
   793→    return <span className="font-mono text-[11px] text-[var(--erp-accent)] font-medium">{formatDocNumber(prefix, Number(value) || sequence)}</span>;
   794→  }
   795→  if (value === undefined || value === null || value === '' || (Array.isArray(value) && value.length === 0)) {
   796→    return <span className="text-[var(--erp-text-muted)]">—</span>;
   797→  }
   798→  if (col.type === 'status') {
   799→    return <Badge variant={statusVariant(String(value))} icon={statusIcon(String(value))}>{String(value)}</Badge>;
   800→  }
   801→  if (col.type === 'priority') {
   802→    return <Badge variant={priorityVariant(String(value))} icon={priorityIcon(String(value))}>{String(value)}</Badge>;
   803→  }
   804→  if (col.type === 'multi_select') {
   805→    const arr = Array.isArray(value) ? value : [value];
   806→    return (
   807→      <div className="flex flex-wrap gap-1">
   808→        {arr.map((v, i) => (
   809→          <span key={i} className="text-[10px] px-1.5 py-0.5 rounded bg-[var(--erp-bg-hover)] text-[var(--erp-text-secondary)]">
   810→            {String(v)}
   811→          </span>
   812→        ))}
   813→      </div>
   814→    );
   815→  }
   816→  if (col.type === 'rating') {
   817→    const n = Number(value) || 0;
   818→    return (
   819→      <span className="text-[var(--erp-warning)] text-[12px]" title={`${n}/5`}>
   820→        {'★'.repeat(n)}<span className="text-[var(--erp-text-muted)]">{'☆'.repeat(5 - n)}</span>
   821→      </span>
   822→    );
   823→  }
   824→  if (col.type === 'currency') {
   825→    const n = Number(value) || 0;
   826→    return <span className="font-mono text-[var(--erp-text)]">{formatCurrencyCompact(n, currency)}</span>;
   827→  }
   828→  if (col.type === 'percentage') {
   829→    const n = Number(value) || 0;
   830→    return (
   831→      <span className="inline-flex items-center gap-1.5">
   832→        <span className="font-mono">{n}%</span>
   833→        <span className="w-12 h-1.5 rounded-full bg-[var(--erp-bg-hover)] overflow-hidden">
   834→          <span className="block h-full rounded-full" style={{ width: `${Math.min(100, n)}%`, background: 'var(--erp-accent)' }} />
   835→        </span>
   836→      </span>
   837→    );
   838→  }
   839→  if (col.type === 'number') {
   840→    return <span className="font-mono">{String(value)}</span>;
   841→  }
   842→  if (col.type === 'date') {
   843→    return <span>{formatDate(String(value))}</span>;
   844→  }
   845→  if (col.type === 'datetime') {
   846→    return <span>{formatDate(String(value))}</span>;
   847→  }
   848→  if (col.type === 'email') {
   849→    return <a href={`mailto:${value}`} className="text-[var(--erp-accent)] hover:underline">{String(value)}</a>;
   850→  }
   851→  if (col.type === 'phone') {
   852→    return <a href={`tel:${value}`} className="text-[var(--erp-accent)] hover:underline">{String(value)}</a>;
   853→  }
   854→  if (col.type === 'long_text') {
   855→    return (
   856→      <div className={cn('text-[var(--erp-text)]', !expanded && 'line-clamp-2 max-w-[260px]')}>
   857→        {String(value)}
   858→      </div>
   859→    );
   860→  }
   861→  return <span className="text-[var(--erp-text)]">{String(value)}</span>;
   862→}

   873→function priorityIcon(priority: string): string {

   875→  if (p === 'critical') return 'fa-triangle-exclamation';
   876→  if (p === 'high') return 'fa-arrow-up';

   882→function guessPrefix(colName: string, registerCode: string): string {
   883→  // Prefer register code (more accurate) — take first 3 letters uppercase
   884→  const codePrefix = registerCode.slice(0, 3).toUpperCase();
   885→  if (codePrefix) return codePrefix;
   886→  const n = colName.toLowerCase();
   887→  if (n.includes('wo')) return 'WO';
   888→  if (n.includes('pm')) return 'PM';
   889→  if (n.includes('cm')) return 'CM';
   890→  return 'REC';
   891→}
