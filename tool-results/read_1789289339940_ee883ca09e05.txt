    53→    53→export function RecordDetailDrawer({ open, register, record, company, onClose, onEdit, onRefresh }: Props) {
    54→    54→  const { hasPermission, user, currency } = useErpStore();
    55→    55→  const [tab, setTab] = useState<Tab>('details');
    56→    56→  const [history, setHistory] = useState<HistoryEntry[]>([]);
    57→    57→  // Initialize as true since data loads on mount (component is keyed by record id)
    58→    58→  const [historyLoading, setHistoryLoading] = useState(true);
    59→    59→  const [related, setRelated] = useState<any[]>([]);
    60→    60→  const [relatedLoading, setRelatedLoading] = useState(false);
    61→    61→  const [workflowOpen, setWorkflowOpen] = useState(false);
    62→    62→  const [editMode, setEditMode] = useState(false);
    63→    63→  const [editData, setEditData] = useState<Record<string, any>>({});
    64→    64→  const [errors, setErrors] = useState<Record<string, string>>({});
    65→    65→  const [saving, setSaving] = useState(false);
    66→    66→  const [masterData, setMasterData] = useState<Record<string, string[]>>({});
    67→    67→
    68→    68→  const canEdit = hasPermission(register.code, 'edit');
    69→    69→  const canApprove = hasPermission(register.code, 'approve');
    70→    70→  const hasStatusCol = register.columns.some((c) => c.type === 'status');
    71→    71→
    72→    72→  // Load master data for dropdowns
    73→    73→  useEffect(() => {
    74→    74→    if (!open) return;
    75→    75→    let cancelled = false;
    76→    76→    masterDataApi.list().then((md) => {
    77→    77→      if (!cancelled) setMasterData(md);
    78→    78→    }).catch(() => {});
    79→    79→    return () => { cancelled = true; };
    80→    80→  }, [open]);
    81→    81→
    82→    82→  // Initialize edit data when entering edit mode
    83→    83→  const enterEditMode = useCallback(() => {
    84→    84→    if (!record) return;
    85→    85→    setEditData({ ...record.data });
    86→    86→    setErrors({});
    87→    87→    setEditMode(true);
    88→    88→  }, [record]);
    89→    89→
    90→    90→  const cancelEdit = useCallback(() => {
    91→    91→    setEditMode(false);
    92→    92→    setEditData({});
    93→    93→    setErrors({});
    94→    94→  }, []);
    95→    95→
    96→    96→  const setField = (name: string, value: any) => {
    97→    97→    setEditData((d) => ({ ...d, [name]: value }));
    98→    98→    setErrors((e) => { const n = { ...e }; delete n[name]; return n; });
    99→    99→  };
   100→   100→
   101→   101→  const saveInlineEdit = async () => {
   102→   102→    if (!record || !register) return;
   103→   103→    const { valid, errors: errs } = validateRecord(editData, register.columns);
   104→   104→    if (!valid) {
   105→   105→      setErrors(errs);
   106→   106→      toast.error('Please fix the errors before saving', {
   107→   107→        description: `${Object.keys(errs).length} field(s) need attention`,
   108→   108→      });
   109→   109→      return;
   110→   110→    }
   111→   111→    setSaving(true);
   112→   112→    try {
   113→   113→      await recordsApi.update(register.id, record.id, editData);
   114→   114→      toast.success('Record updated successfully');
   115→   115→      setEditMode(false);
   116→   116→      onRefresh();
   117→   117→    } catch (e: any) {
   118→   118→      toast.error('Failed to save record', { description: e.message });
   119→   119→    } finally {
   120→   120→      setSaving(false);
   121→   121→    }
   122→   122→  };
   123→   123→
   124→   124→  // Load history when drawer opens or record changes
   125→   125→  useEffect(() => {
   126→   126→    if (!open || !record || !register) return;
   127→   127→    let cancelled = false;
   128→   128→    recordsApi.getHistory(register.id, record.id).then((res) => {
   129→   129→      if (cancelled) return;
   130→   130→      setHistory(res.history || []);
   131→   131→    }).catch((e) => {
   132→   132→      if (cancelled) return;
   133→   133→      console.error('Failed to load history', e);
   134→   134→    }).finally(() => {
   135→   135→      if (cancelled) return;
   136→   136→      setHistoryLoading(false);
   137→   137→    });
   138→   138→    return () => { cancelled = true; };
   139→   139→  }, [open, record, register.id]);
   140→   140→
   141→   141→  // Load related records when the Related tab is clicked
   142→   142→  useEffect(() => {
   143→   143→    if (tab !== 'related' || !record) return;
   144→   144→    let cancelled = false;
   145→   145→    setRelatedLoading(true);
   146→   146→    const relatedPromise = register ? recordsApi.getRelated(register.id, record.id) : Promise.resolve({ related: [] });
   147→   147→    relatedPromise.then((res) => {
   148→   148→      if (cancelled) return;
   149→   149→      setRelated(res.related || []);
   150→   150→    }).catch((e) => {
   151→   151→      if (cancelled) return;
   152→   152→      console.error('Failed to load related records', e);
   153→   153→    }).finally(() => {
   154→   154→      if (cancelled) return;
   155→   155→      setRelatedLoading(false);
   156→   156→    });
   157→   157→    return () => { cancelled = true; };
   158→   158→  }, [tab, record, register.id]);
   159→   159→
   160→   160→  // Close on Escape
   161→   161→  useEffect(() => {
   162→   162→    if (!open) return;
   163→   163→    const handler = (e: KeyboardEvent) => {
   164→   164→      if (e.key === 'Escape' && !workflowOpen) {
   165→   165→        if (editMode) {
   166→   166→          cancelEdit();
   167→   167→        } else {
   168→   168→          onClose();
   169→   169→        }
   170→   170→      }
   171→   171→    };
   172→   172→    window.addEventListener('keydown', handler);
   173→   173→    return () => window.removeEventListener('keydown', handler);
   174→   174→  }, [open, workflowOpen, onClose, editMode, cancelEdit]);
   175→   175→
   176→   176→  // Exit edit mode when drawer closes
   177→   177→  useEffect(() => {
   178→   178→    if (!open && editMode) {
   179→   179→      setEditMode(false);
   180→   180→      setEditData({});
   181→   181→      setErrors({});
   182→   182→    }
   183→   183→  }, [open, editMode]);
   184→   184→
   185→   185→  const statusCol = register ? register.columns.find((c) => c.type === 'status') : null;
   186→   186→  const currentStatus = statusCol && record ? String(record.data[statusCol.name] || '—') : null;
   187→   187→
   188→   188→  const handlePrint = () => {
   189→   189→    if (!register || !record) return;
   190→   190→    if (register && record) printRecord(register, record, company, currency);
   191→   191→  };
   192→   192→
   193→   193→  const handleWorkflowClick = () => {
   194→   194→    setWorkflowOpen(true);
   195→   195→  };
   196→   196→
   197→   197→  const handleWorkflowTransition = () => {
   198→   198→    setWorkflowOpen(false);
   199→   199→    onRefresh();
   200→   200→  };
   201→   201→
   202→   202→  // Don't render if no record (drawer closed)
   203→   203→  if (!record) return null;
   204→   204→
   205→   205→  return (
   206→   206→    <Dialog open={open} onOpenChange={(o) => { if (!o) { setEditMode(false); onClose(); } }}>
   207→   207→      <DialogContent className="max-w-[600px] w-full h-[90vh] max-h-[90vh] flex flex-col p-0 gap-0">
   208→   208→        {/* Header */}
   209→   209→        <div className="flex items-center gap-3 px-4 h-[60px] border-b border-[var(--erp-border)] bg-[var(--erp-bg-card)] shrink-0">
   210→   210→          <div
   211→   211→            className="w-9 h-9 rounded-lg flex items-center justify-center shrink-0"
   212→   212→            style={{ background: `linear-gradient(135deg, ${register.color}30, ${register.color}10)`, color: register.color }}
   213→   213→          >
   214→   214→            <FAIcon name={register.icon} className="text-[14px]" />
   215→   215→          </div>
   216→   216→          <div className="flex-1 min-w-0">
   217→   217→            <div className="text-[14px] font-semibold text-[var(--erp-text)] truncate" style={{ fontFamily: 'var(--font-display)' }}>
   218→   218→              {register.name} — #{record.sequence}
   219→   219→            </div>
   220→   220→            <div className="text-[10px] text-[var(--erp-text-muted)] flex items-center gap-1.5">
   221→   221→              <span>Record ID: {record.id.slice(-8)}</span>
   222→   222→              {currentStatus && (
   223→   223→                <>
   224→   224→                  <span>·</span>
   225→   225→                  <StatusPill status={currentStatus} />
   226→   226→                </>
   227→   227→              )}
   228→   228→            </div>
   229→   229→          </div>
   230→   230→          {editMode && (
   231→   231→            <div className="flex items-center gap-1 mr-2">
   232→   232→              <span className="text-[10px] px-2 py-0.5 rounded-full bg-[var(--erp-accent-dim)] text-[var(--erp-accent)] font-semibold flex items-center gap-1">
   233→   233→                <Pencil className="w-2.5 h-2.5" /> EDITING
   234→   234→              </span>
   235→   235→            </div>
   236→   236→          )}
   237→   237→          <button
   238→   238→            type="button"
   239→   239→            onClick={() => { setEditMode(false); onClose(); }}
   240→   240→            className="p-2.5 rounded-lg bg-[var(--erp-bg-hover)] text-[var(--erp-text)] hover:bg-[var(--erp-danger)] hover:text-white transition-all shrink-0"
   241→   241→            aria-label="Close"
   242→   242→          >
   243→   243→            <X className="w-4 h-4" />
   244→   244→          </button>
   245→   245→        </div>
   246→   246→        {/* Action bar */}
   247→   247→        <div className="flex items-center gap-1.5 px-4 py-2 border-b border-[var(--erp-border)] bg-[var(--erp-bg-card)] shrink-0">
   248→   248→          {!editMode ? (
   249→   249→            <>
   250→   250→              <Button variant="outline" size="sm" onClick={handlePrint} className="h-8 text-[11px]">
   251→   251→                <Printer className="w-3.5 h-3.5 mr-1" /> Print
   252→   252→              </Button>
   253→   253→              {hasStatusCol && (canApprove || canEdit) && (
   254→   254→                <Button variant="outline" size="sm" onClick={handleWorkflowClick} className="h-8 text-[11px] border-[var(--erp-accent-border)] text-[var(--erp-accent)] hover:bg-[var(--erp-accent-dim)]">
   255→   255→                  <WorkflowIcon className="w-3.5 h-3.5 mr-1" /> Workflow
   256→   256→                </Button>
   257→   257→              )}
   258→   258→              {canEdit && (
   259→   259→                <Button size="sm" onClick={enterEditMode} className="h-8 text-[11px] ml-auto bg-[var(--erp-accent)] hover:bg-[var(--erp-accent-hover)]">
   260→   260→                  <Pencil className="w-3.5 h-3.5 mr-1" /> Inline Edit
   261→   261→                </Button>
   262→   262→              )}
   263→   263→            </>
   264→   264→          ) : (
   265→   265→            <>
   266→   266→              <Button variant="outline" size="sm" onClick={cancelEdit} disabled={saving} className="h-8 text-[11px]">
   267→   267→                <XCircle className="w-3.5 h-3.5 mr-1" /> Cancel
   268→   268→              </Button>
   269→   269→              <Button size="sm" onClick={saveInlineEdit} disabled={saving} className="h-8 text-[11px] ml-auto bg-[var(--erp-success)] hover:bg-[var(--erp-success)]/90">
   270→   270→                {saving ? <Loader2 className="w-3.5 h-3.5 mr-1 animate-spin" /> : <Check className="w-3.5 h-3.5 mr-1" />}
   271→   271→                {saving ? 'Saving...' : 'Save Changes'}
   272→   272→              </Button>
   273→   273→            </>
   274→   274→          )}
   275→   275→        </div>
   276→   276→
   277→   277→        {/* Error banner (if validation errors) */}
   278→   278→        {editMode && Object.keys(errors).length > 0 && (
   279→   279→          <div className="px-4 py-2 border-b border-[var(--erp-danger)]/30 bg-[rgba(239,68,68,0.05)] flex items-center gap-2 text-[11px] text-[var(--erp-danger)]">
   280→   280→            <AlertCircle className="w-3.5 h-3.5 shrink-0" />
   281→   281→            <span>{Object.keys(errors).length} field(s) need attention</span>
   282→   282→          </div>
   283→   283→        )}
   284→   284→
   285→   285→        {/* Tabs (hidden in edit mode to maximize space) */}
   286→   286→        {!editMode && (
   287→   287→          <div className="flex items-center gap-1 px-4 py-1.5 border-b border-[var(--erp-border)] bg-[var(--erp-bg-card)] shrink-0">
   288→   288→            <TabButton active={tab === 'details'} onClick={() => setTab('details')} icon={<FileText className="w-3.5 h-3.5" />} label="Details" />
   289→   289→            <TabButton active={tab === 'history'} onClick={() => setTab('history')} icon={<HistoryIcon className="w-3.5 h-3.5" />} label="History" count={history.length} />
   290→   290→            <TabButton active={tab === 'related'} onClick={() => setTab('related')} icon={<Link2 className="w-3.5 h-3.5" />} label="Related" />
   291→   291→            <TabButton active={tab === 'activity'} onClick={() => setTab('activity')} icon={<Activity className="w-3.5 h-3.5" />} label="Activity" />
   292→   292→          </div>
   293→   293→        )}
   294→   294→
   295→   295→        {/* Content */}
   296→   296→        <div className="flex-1 overflow-y-auto bg-[var(--erp-bg)]">
   297→   297→          {editMode ? (
   298→   298→            <InlineEditTab
   299→   299→              register={register}
   300→   300→              data={editData}
   301→   301→              errors={errors}
   302→   302→              masterData={masterData}
   303→   303→              onChange={setField}
   304→   304→              currency={currency}
   305→   305→            />
   306→   306→          ) : (
   307→   307→            <>
   308→   308→              {tab === 'details' && (
   309→   309→                <DetailsTab register={register} record={record} currency={currency} onRefresh={onRefresh} />
   310→   310→              )}
   311→   311→              {tab === 'history' && (
   312→   312→                <HistoryTab history={history} loading={historyLoading} hasStatusCol={!!hasStatusCol} />
   313→   313→              )}
   314→   314→              {tab === 'related' && (
   315→   315→                <RelatedTab related={related} loading={relatedLoading} />
   316→   316→              )}
   317→   317→              {tab === 'activity' && (
   318→   318→                <ActivityTab record={record} register={register} />
   319→   319→              )}
   320→   320→            </>
   321→   321→          )}
   322→   322→        </div>
   323→   323→
   324→   324→        {/* Footer with CLOSE button */}
   325→   325→        <div className="px-4 py-2 border-t border-[var(--erp-border)] bg-[var(--erp-bg-card)] shrink-0 flex items-center justify-between text-[10px] text-[var(--erp-text-muted)]">
   326→   326→          <span className="flex items-center gap-1">
   327→   327→            <Clock className="w-3 h-3" />
   328→   328→            Created {formatDate(record.createdAt)}
   329→   329→          </span>
   330→   330→          <span>Updated {formatTimeAgo(record.updatedAt)}</span>
   331→   331→        </div>
   332→   332→        {/* Big visible CLOSE bar — always works */}
   333→   333→        <button
   334→   334→          type="button"
   335→   335→          onClick={() => { setEditMode(false); onClose(); }}
   336→   336→          className="w-full py-3 bg-[var(--erp-accent)] text-white text-[12px] font-semibold hover:bg-[var(--erp-accent-hover)] transition-colors shrink-0"
   337→   337→        >
   338→   338→          ✕ Close
   339→   339→        </button>
   340→   340→
   341→   341→        {/* Workflow modal */}
   342→   342→        <ApprovalWorkflow
   343→   343→          open={workflowOpen}
   344→   344→          register={register}
   345→   345→          record={record}
   346→   346→          onClose={() => setWorkflowOpen(false)}
   347→   347→          onTransition={handleWorkflowTransition}
   348→   348→        />
   349→   349→      </DialogContent>
   350→   350→    </Dialog>
   351→   351→  );
   352→   352→}

   355→   355→function InlineEditTab({
   356→   356→  register, data, errors, masterData, onChange, currency = 'AED',
   357→   357→}: {
   358→   358→  register: Register;
   359→   359→  data: Record<string, any>;
   360→   360→  errors: Record<string, string>;
   361→   361→  masterData: Record<string, string[]>;
   362→   362→  onChange: (name: string, value: any) => void;
   363→   363→  currency?: string;
   364→   364→}) {
   365→   365→  const cols = register.columns.filter((c) => c.type !== 'auto_increment');
   366→   366→
   367→   367→  // Group columns by type for better layout (same as DetailsTab)
   368→   368→  const mainFields = cols.filter((c) => ['text', 'date', 'datetime', 'time', 'number', 'currency', 'percentage', 'email', 'phone', 'dropdown', 'status', 'priority', 'rating', 'employee', 'department', 'building', 'asset', 'equipment', 'vendor', 'url', 'color'].includes(c.type));
   369→   369→  const longFields = cols.filter((c) => c.type === 'long_text');
   370→   370→  const multiFields = cols.filter((c) => c.type === 'multi_select' || c.type === 'tags');
   371→   371→  const imageFields = cols.filter((c) => c.type === 'image');
   372→   372→
   373→   373→  return (
   374→   374→    <div className="p-4 space-y-5">
   375→   375→      <div className="flex items-center gap-2 p-2.5 rounded-md bg-[var(--erp-accent-dim)] border border-[var(--erp-accent-border)] text-[11px] text-[var(--erp-accent)]">
   376→   376→        <Pencil className="w-3.5 h-3.5 shrink-0" />
   377→   377→        <span>Edit fields directly. Changes are saved when you click "Save Changes".</span>
   378→   378→      </div>
   379→   379→
   380→   380→      {/* Image gallery (inline-edit) */}
   381→   381→      {imageFields.length > 0 && (
   382→   382→        <div>
   383→   383→          <SectionLabel icon="fa-image" label="Images" count={imageFields.length} />
   384→   384→          <div className="grid grid-cols-2 sm:grid-cols-3 gap-2.5">
   385→   385→            {imageFields.map((col) => (
   386→   386→              <InlineField key={col.name} col={col} value={data[col.name]} error={errors[col.name]} masterData={masterData} onChange={(v) => onChange(col.name, v)} currency={currency} />
   387→   387→            ))}
   388→   388→          </div>
   389→   389→        </div>
   390→   390→      )}
   391→   391→
   392→   392→      {/* Main fields grid */}
   393→   393→      <div>
   394→   394→        <SectionLabel icon="fa-circle-info" label="Record Fields" count={mainFields.length} />
   395→   395→        <div className="grid grid-cols-2 gap-2.5">
   396→   396→          {mainFields.map((col) => (
   397→   397→            <InlineField key={col.name} col={col} value={data[col.name]} error={errors[col.name]} masterData={masterData} onChange={(v) => onChange(col.name, v)} currency={currency} />
   398→   398→          ))}
   399→   399→        </div>
   400→   400→      </div>
   401→   401→
   402→   402→      {/* Multi-select fields */}
   403→   403→      {multiFields.length > 0 && (
   404→   404→        <div>
   405→   405→          <SectionLabel icon="fa-list-check" label="Tags & Categories" count={multiFields.length} />
   406→   406→          <div className="space-y-2.5">
   407→   407→            {multiFields.map((col) => (
   408→   408→              <InlineField key={col.name} col={col} value={data[col.name]} error={errors[col.name]} masterData={masterData} onChange={(v) => onChange(col.name, v)} fullWidth currency={currency} />
   409→   409→            ))}
   410→   410→          </div>
   411→   411→        </div>
   412→   412→      )}
   413→   413→
   414→   414→      {/* Long text fields */}

   721→   721→function MultiSelectInline({ options, value, onChange }: { options: string[]; value: string[]; onChange: (v: string[]) => void }) {
   722→   722→  const toggle = (opt: string) => {
   723→   723→    if (value.includes(opt)) onChange(value.filter((v) => v !== opt));
   724→   724→    else onChange([...value, opt]);
   725→   725→  };
   726→   726→  return (
   727→   727→    <div className="flex flex-wrap gap-1.5 min-h-[32px]">
   728→   728→      {options.length === 0 ? (
   729→   729→        <span className="text-[11px] text-[var(--erp-text-muted)]">No options available</span>
   730→   730→      ) : (
   731→   731→        options.map((opt) => {
   732→   732→          const selected = value.includes(opt);
   733→   733→          return (
   734→   734→            <button
   735→   735→              key={opt}
   736→   736→              type="button"
   737→   737→              onClick={() => toggle(opt)}
   738→   738→              className={cn(
   739→   739→                'text-[11px] px-2 py-1 rounded-md transition-colors',
   740→   740→                selected
   741→   741→                  ? 'bg-[var(--erp-accent)] text-white'
   742→   742→                  : 'bg-[var(--erp-bg-card)] text-[var(--erp-text-secondary)] hover:bg-[var(--erp-bg-hover)]',
   743→   743→              )}
   744→   744→            >
   745→   745→              {selected && '✓ '}
   746→   746→              {opt}
   747→   747→            </button>
   748→   748→          );
   749→   749→        })
   750→   750→      )}
   751→   751→    </div>
   752→   752→  );
   753→   753→}

   755→   755→// Image upload field for the drawer's inline edit mode
   756→   756→function DrawerImageField({ value, onChange }: { value: any; onChange: (v: any) => void }) {
   757→   757→  const [uploading, setUploading] = useState(false);
   758→   758→  const fileInputRef = useRef<HTMLInputElement>(null);
   759→   759→
   760→   760→  const handleFile = async (file: File) => {
   761→   761→    if (!file) return;
   762→   762→    setUploading(true);
   763→   763→    try {
   764→   764→      const result = await uploadsApi.upload(file);
   765→   765→      onChange(result.url);
   766→   766→      toast.success('Image uploaded');
   767→   767→    } catch (e: any) {
   768→   768→      toast.error('Upload failed', { description: e.message });
   769→   769→    } finally {
   770→   770→      setUploading(false);
   771→   771→    }
   772→   772→  };
   773→   773→
   774→   774→  return (
   775→   775→    <div className="flex items-center gap-2">
   776→   776→      <input
   777→   777→        ref={fileInputRef}
   778→   778→        type="file"
   779→   779→        accept="image/*"
   780→   780→        className="hidden"
   781→   781→        onChange={(e) => { const f = e.target.files?.[0]; if (f) handleFile(f); e.target.value = ''; }}
   782→   782→      />
   783→   783→      {value ? (
   784→   784→        <div className="relative">
   785→   785→          <img src={String(value)} alt="preview" className="w-16 h-16 object-cover rounded-md border border-[var(--erp-border)]" />
   786→   786→          <button
   787→   787→            type="button"
   788→   788→            onClick={() => onChange('')}
   789→   789→            className="absolute -top-2 -right-2 w-5 h-5 rounded-full bg-[var(--erp-danger)] text-white flex items-center justify-center hover:scale-110 transition-transform"
   790→   790→            aria-label="Remove image"
   791→   791→          >
   792→   792→            <X className="w-3 h-3" />
   793→   793→          </button>
   794→   794→        </div>
   795→   795→      ) : (
   796→   796→        <div className="w-16 h-16 rounded-md border border-dashed border-[var(--erp-border)] flex items-center justify-center text-[var(--erp-text-muted)]">
   797→   797→          {uploading ? <Loader2 className="w-4 h-4 animate-spin" /> : <Upload className="w-4 h-4" />}
   798→   798→        </div>
   799→   799→      )}
   800→   800→      <Button
   801→   801→        type="button"
   802→   802→        variant="outline"
   803→   803→        size="sm"
   804→   804→        onClick={() => fileInputRef.current?.click()}
   805→   805→        disabled={uploading}
   806→   806→        className="h-8 text-[11px]"
   807→   807→      >
   808→   808→        {uploading ? <><Loader2 className="w-3.5 h-3.5 mr-1 animate-spin" /> Uploading...</> : <><Upload className="w-3.5 h-3.5 mr-1" /> {value ? 'Replace' : 'Upload'}</>}
   809→   809→      </Button>
   810→   810→    </div>
   811→   811→  );
   812→   812→}
