     1→'use client';
     2→
     3→// FMCore ERP — Settings view (with Saved Views management tab)
     4→import { useEffect, useState } from 'react';
     5→import { settingsApi, backupApi, savedViewsApi, statsApi, type SavedViewMeta, type SystemStats } from '@/lib/erp/api';
     6→import type { Setting } from '@/lib/erp/types';
     7→import { useErpStore } from '@/lib/erp/store';
     8→import { FAIcon } from './icon';
     9→import { EmptyStateIllustration } from './empty-state-illustration';
    10→import {
    11→  Download, Upload, RotateCcw, Save, Building2, Palette, FileText,
    12→  Hash, Bell, Database, Shield, Info, Bookmark, Trash2, Globe, Lock, Pencil,
    13→  Search, Filter as FilterIcon, ArrowUpDown, Check, X, Loader2, Rocket, AlertCircle,
    14→} from 'lucide-react';
    15→import { Button } from '@/components/ui/button';
    16→import { Input } from '@/components/ui/input';
    17→import { Label } from '@/components/ui/label';
    18→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
    19→import { toast } from 'sonner';
    20→import { cn } from '@/lib/utils';
    21→import { formatTimeAgo } from '@/lib/erp/utils';
    22→import { ProjectStatusPanel } from './project-status-panel';
    23→import { DeploymentGuide } from './deployment-guide';
    24→
    25→export function SettingsView() {
    26→  const { theme, setTheme, currency, setCurrency, rtl, setRtl, user, hasPermission } = useErpStore();
    27→  const [settings, setSettings] = useState<Record<string, string>>({});
    28→  const [saving, setSaving] = useState(false);
    29→  const [activeTab, setActiveTab] = useState('company');
    30→  const [savedViews, setSavedViews] = useState<SavedViewMeta[]>([]);
    31→  const [viewsLoading, setViewsLoading] = useState(false);
    32→  const [viewSearch, setViewSearch] = useState('');
    33→  const [editTarget, setEditTarget] = useState<SavedViewMeta | null>(null);
    34→  const [editName, setEditName] = useState('');
    35→  const [editShared, setEditShared] = useState(false);
    36→  const [editSaving, setEditSaving] = useState(false);
    37→  const [systemStats, setSystemStats] = useState<SystemStats | null>(null);
    38→
    39→  useEffect(() => {
    40→    settingsApi.list().then((list: Setting[]) => {
    41→      const map: Record<string, string> = {};
    42→      list.forEach((s) => (map[s.key] = s.value));
    43→      setSettings(map);
    44→    }).catch(() => {});
    45→  }, []);
    46→
    47→  const loadSavedViews = () => {
    48→    setViewsLoading(true);
    49→    savedViewsApi.listAll().then((views) => {
    50→      setSavedViews(views);
    51→    }).catch(() => {}).finally(() => setViewsLoading(false));
    52→  };
    53→
    54→  useEffect(() => {
    55→    if (activeTab === 'saved-views') loadSavedViews();
    56→    if (activeTab === 'about') {
    57→      statsApi.get().then(setSystemStats).catch(() => {});
    58→    }
    59→  }, [activeTab]);
    60→
    61→  const handleDeleteView = async (id: string, name: string) => {
    62→    try {
    63→      await savedViewsApi.remove(id);
    64→      toast.success(`Deleted view "${name}"`);
    65→      setSavedViews((v) => v.filter((x) => x.id !== id));
    66→    } catch (e: any) {
    67→      toast.error('Failed to delete view', { description: e.message });
    68→    }
    69→  };
    70→
    71→  const handleEditView = (view: SavedViewMeta) => {
    72→    setEditTarget(view);
    73→    setEditName(view.name);
    74→    setEditShared(view.isShared);
    75→  };
    76→
    77→  const handleSaveEdit = async () => {
    78→    if (!editTarget) return;
    79→    if (!editName.trim()) {
    80→      toast.error('View name cannot be empty');
    81→      return;
    82→    }
    83→    setEditSaving(true);
    84→    try {
    85→      await savedViewsApi.update({
    86→        id: editTarget.id,
    87→        name: editName.trim(),
    88→        isShared: editShared,
    89→      });
    90→      toast.success(`Updated view "${editName}"`);
    91→      setEditTarget(null);
    92→      setEditName('');
    93→      setEditShared(false);
    94→      loadSavedViews();
    95→    } catch (e: any) {
    96→      toast.error('Failed to update view', { description: e.message });
    97→    } finally {
    98→      setEditSaving(false);
    99→    }
   100→  };
   101→
   102→  useEffect(() => {
   103→    settingsApi.list().then((list: Setting[]) => {
   104→      const map: Record<string, string> = {};
   105→      list.forEach((s) => (map[s.key] = s.value));
   106→      setSettings(map);
   107→    }).catch(() => {});
   108→  }, []);
   109→
   110→  const update = (key: string, value: string) => {
   111→    setSettings((s) => ({ ...s, [key]: value }));
   112→  };
   113→
   114→  const saveAll = async () => {
   115→    setSaving(true);
   116→    try {
   117→      const items = Object.entries(settings).map(([key, value]) => ({ key, value }));
   118→      await settingsApi.bulkSet(items);
   119→      toast.success(`${items.length} settings saved`);
   120→    } catch (e: any) {
   121→      toast.error('Failed to save settings', { description: e.message });
   122→    } finally {
   123→      setSaving(false);
   124→    }
   125→  };
   126→
   127→  const exportBackup = async () => {
   128→    try {
   129→      const backup = await backupApi.export();
   130→      const blob = new Blob([JSON.stringify(backup, null, 2)], { type: 'application/json' });
   131→      const url = URL.createObjectURL(blob);
   132→      const a = document.createElement('a');
   133→      a.href = url;
   134→      a.download = `fmcore-backup-${new Date().toISOString().slice(0, 10)}.json`;
   135→      a.click();
   136→      URL.revokeObjectURL(url);
   137→      toast.success('Backup exported');
   138→    } catch (e: any) {
   139→      toast.error('Backup failed', { description: e.message });
   140→    }
   141→  };
   142→
   143→  const importBackup = () => {
   144→    const input = document.createElement('input');
   145→    input.type = 'file';
   146→    input.accept = 'application/json';
   147→    input.onchange = async (e) => {
   148→      const file = (e.target as HTMLInputElement).files?.[0];
   149→      if (!file) return;
   150→      if (!confirm('Importing will REPLACE all current data. Continue?')) return;
   151→      try {
   152→        const text = await file.text();
   153→        const data = JSON.parse(text);
   154→        await backupApi.import(data);
   155→        toast.success('Backup imported successfully');
   156→        setTimeout(() => window.location.reload(), 800);
   157→      } catch (e: any) {
   158→        toast.error('Import failed', { description: e.message });
   159→      }
   160→    };
   161→    input.click();
   162→  };
   163→
   164→  const resetDb = async () => {
   165→    if (!confirm('This will erase ALL data and re-seed sample data. Continue?')) return;
   166→    try {
   167→      const res = await fetch('/api/erp/reset', { method: 'POST' });
   168→      const data = await res.json();
   169→      toast.success(`Database reset — ${data.registers} registers, ${data.records} records`);
   170→      setTimeout(() => window.location.reload(), 800);
   171→    } catch (e: any) {
   172→      toast.error('Reset failed', { description: e.message });
   173→    }
   174→  };
   175→
   176→  // Build TABS array — restrict sensitive tabs to admins only
   177→  const canEditSettings = hasPermission('settings', 'edit');
   178→  const TABS = [
   179→    { id: 'company',     label: 'Company',        icon: <Building2 className="w-4 h-4" /> },
   180→    { id: 'appearance',  label: 'Appearance',    icon: <Palette className="w-4 h-4" /> },
   181→    { id: 'numbering',   label: 'Document #',    icon: <Hash className="w-4 h-4" /> },
   182→    { id: 'notifications', label: 'Notifications', icon: <Bell className="w-4 h-4" /> },
   183→    { id: 'deploy',       label: 'Deploy Guide',   icon: <Rocket className="w-4 h-4" /> },
   184→    ...(canEditSettings ? [{ id: 'saved-views', label: 'Saved Views', icon: <Bookmark className="w-4 h-4" /> }] : []),
   185→    ...(canEditSettings ? [{ id: 'backup', label: 'Backup & Reset', icon: <Database className="w-4 h-4" /> }] : []),
   186→    { id: 'project',     label: 'Project Status', icon: <Rocket className="w-4 h-4" /> },
   187→    { id: 'about',       label: 'About',         icon: <Info className="w-4 h-4" /> },
   188→  ];
   189→
   190→  return (
   191→    <div className="flex flex-col h-full">
   192→      {/* Header */}
   193→      <div className="px-4 md:px-6 py-3 border-b border-[var(--erp-border)] bg-[var(--erp-bg-card)] flex items-center justify-between">
   194→        <div className="flex items-center gap-3">
   195→          <div className="w-10 h-10 rounded-lg flex items-center justify-center bg-[var(--erp-bg-hover)] text-[var(--erp-text-secondary)]">
   196→            <FAIcon name="fa-cog" />
   197→          </div>

   204→          <Button onClick={saveAll} disabled={saving} className="h-9 text-[12px] bg-[var(--erp-accent)] hover:bg-[var(--erp-accent-hover)]">
   205→            <Save className="w-4 h-4 mr-1" /> {saving ? 'Saving...' : 'Save Changes'}
   206→          </Button>

   209→            Read-only access
   210→          </span>
   211→        )}
   212→      </div>
   213→
   214→      <div className="flex flex-1 overflow-hidden">
   215→        {/* Sub-tabs */}
   216→        <nav className="w-[180px] border-r border-[var(--erp-border)] bg-[var(--erp-bg-secondary)] p-2 hidden md:block shrink-0">
   217→          {TABS.map((t) => (
   218→            <button
   219→              key={t.id}
   220→              onClick={() => setActiveTab(t.id)}
   221→              className={`flex items-center gap-2 w-full px-3 py-2 rounded-md text-[12px] transition-colors ${
   222→                activeTab === t.id
   223→                  ? 'bg-[var(--erp-accent-dim)] text-[var(--erp-accent)] font-medium'
   224→                  : 'text-[var(--erp-text-secondary)] hover:bg-[var(--erp-bg-hover)]'
   225→              }`}
   226→            >
   227→              {t.icon}
   228→              <span>{t.label}</span>
   229→            </button>
   230→          ))}
   231→        </nav>
   232→
   233→        {/* Content */}
   234→        <div className="flex-1 overflow-y-auto p-4 md:p-6">
   235→          {/* Mobile tabs */}
   236→          <div className="md:hidden flex gap-1 mb-4 overflow-x-auto no-scrollbar">
   237→            {TABS.map((t) => (
   238→              <button
   239→                key={t.id}
   240→                onClick={() => setActiveTab(t.id)}
   241→                className={`px-3 py-1.5 rounded-md text-[11px] whitespace-nowrap ${
   242→                  activeTab === t.id ? 'bg-[var(--erp-accent)] text-white' : 'bg-[var(--erp-bg-card)] text-[var(--erp-text-secondary)]'
   243→                }`}
   244→              >
   245→                {t.label}
   246→              </button>
   247→            ))}
   248→          </div>
   249→
   250→          {activeTab === 'company' && (
   251→            <div className="space-y-4 max-w-2xl">
   252→              <h2 className="text-[15px] font-semibold text-[var(--erp-text)] flex items-center gap-2">
   253→                <Building2 className="w-4 h-4 text-[var(--erp-accent)]" /> Company Profile
   254→              </h2>
   255→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   256→                <Field label="Company Name" value={settings['company.name'] || ''} onChange={(v) => update('company.name', v)} disabled={!canEditSettings} />
   257→                <Field label="Tax Number (TRN)" value={settings['company.tax_number'] || ''} onChange={(v) => update('company.tax_number', v)} disabled={!canEditSettings} />
   258→                <Field label="Phone" value={settings['company.phone'] || ''} onChange={(v) => update('company.phone', v)} disabled={!canEditSettings} />
   259→                <Field label="Email" value={settings['company.email'] || ''} onChange={(v) => update('company.email', v)} disabled={!canEditSettings} />
   260→                <Field label="Address" value={settings['company.address'] || ''} onChange={(v) => update('company.address', v)} fullWidth disabled={!canEditSettings} />
   261→                <Field label="Country" value={settings['company.country'] || ''} onChange={(v) => update('company.country', v)} disabled={!canEditSettings} />
   262→                <Field label="Fiscal Year" value={settings['company.fiscal_year'] || ''} onChange={(v) => update('company.fiscal_year', v)} disabled={!canEditSettings} />
   263→                <div>
   264→                  <Label className="text-[11px] mb-1">Currency</Label>
   265→                  <div className="flex items-center gap-2 flex-wrap">
   266→                    <Select value={settings['company.currency'] || currency} onValueChange={(v) => { update('company.currency', v); update('currency', v); setCurrency(v); }}>
   267→                      <SelectTrigger className="h-9 text-[12px] bg-[var(--erp-bg-input)] w-[110px]"><SelectValue /></SelectTrigger>
   268→                      <SelectContent>
   269→                        {['AED', 'USD', 'EUR', 'GBP', 'PKR', 'SAR', 'QAR', 'KWD', 'BHD', 'OMR', 'INR', 'JPY', 'CNY', 'CHF', 'CAD', 'AUD', 'Custom'].map((c) => (
   270→                          <SelectItem key={c} value={c} className="text-[12px]">{c}</SelectItem>
   271→                        ))}
   272→                      </SelectContent>
   273→                    </Select>
   274→                    {(settings['company.currency'] === 'Custom' || currency === 'Custom') && (
   275→                      <Input
   276→                        value={settings['company.currency_custom'] || ''}
   277→                        onChange={(e) => { update('company.currency_custom', e.target.value.toUpperCase()); setCurrency(e.target.value.toUpperCase()); }}
   278→                        className="h-9 text-[12px] bg-[var(--erp-bg-input)] w-[80px] uppercase font-mono"
   279→                        placeholder="e.g. BHD"
   280→                        maxLength={5}
   281→                      />
   282→                    )}
   283→                    <span className="text-[10px] text-[var(--erp-text-muted)] flex items-center gap-1">
   284→                      <span className="inline-block w-3 h-3 rounded-full border border-[var(--erp-border)]" style={{ background: 'var(--erp-accent)' }} />
   285→                      Active: <span className="font-mono font-semibold text-[var(--erp-text)]">{settings['company.currency'] === 'Custom' ? (settings['company.currency_custom'] || '—') : (settings['company.currency'] || currency)}</span>
   286→                    </span>
   287→                  </div>
   288→                  <p className="text-[10px] text-[var(--erp-text-muted)] mt-1.5">
   289→                    Applies to all currency fields across registers, dashboards, forms, and printed documents.
   290→                    {settings['company.currency'] === 'Custom' && ' Tip: Common Gulf currencies (QAR, KWD, BHD, OMR) are now in the dropdown.'}
   291→                  </p>
   292→                </div>
   293→              </div>
   294→            </div>
   295→          )}
   296→
   297→          {activeTab === 'appearance' && (
   298→            <div className="space-y-4 max-w-2xl">
   299→              <h2 className="text-[15px] font-semibold text-[var(--erp-text)] flex items-center gap-2">
   300→                <Palette className="w-4 h-4 text-[var(--erp-accent)]" /> Appearance
   301→              </h2>
   302→              <div className="grid grid-cols-2 gap-3">
   303→                {(['dark', 'light'] as const).map((t) => (
   304→                  <button
   305→                    key={t}
   306→                    onClick={() => { setTheme(t); update('theme', t); }}
   307→                    className={`p-4 rounded-lg border-2 text-left transition-all ${
   308→                      theme === t
   309→                        ? 'border-[var(--erp-accent)] bg-[var(--erp-accent-dim)]'
   310→                        : 'border-[var(--erp-border)] hover:border-[var(--erp-text-muted)]'
   311→                    }`}
   312→                  >
   313→                    <div className="flex items-center gap-2 mb-2">
   314→                      <FAIcon name={t === 'dark' ? 'fa-moon' : 'fa-sun'} className="text-[16px]" />
   315→                      <span className="text-[13px] font-medium capitalize text-[var(--erp-text)]">{t} Mode</span>
   316→                    </div>
   317→                    <div className="text-[11px] text-[var(--erp-text-muted)]">
   318→                      {t === 'dark' ? 'Easy on the eyes, great for low-light environments' : 'Clean and bright, ideal for daytime use'}
   319→                    </div>
   320→                  </button>
   321→                ))}
   322→              </div>
   323→              <div>
   324→                <Label className="text-[11px] mb-1">Date Format</Label>
   325→                <Select value={settings['date_format'] || 'DD/MM/YYYY'} onValueChange={(v) => update('date_format', v)}>
   326→                  <SelectTrigger className="h-9 text-[12px] bg-[var(--erp-bg-input)] w-[200px]"><SelectValue /></SelectTrigger>
   327→                  <SelectContent>
   328→                    {['DD/MM/YYYY', 'MM/DD/YYYY', 'YYYY-MM-DD'].map((c) => (
   329→                      <SelectItem key={c} value={c} className="text-[12px]">{c}</SelectItem>
   330→                    ))}
   331→                  </SelectContent>
   332→                </Select>
   333→              </div>
   334→              {/* RTL Toggle */}
   335→              <div className="flex items-center gap-3 p-3 rounded-md border border-[var(--erp-border)] bg-[var(--erp-bg-input)]">
   336→                <div className="flex-1">
   337→                  <div className="text-[12px] font-medium text-[var(--erp-text)]">Right-to-Left (RTL) Layout</div>
   338→                  <div className="text-[10px] text-[var(--erp-text-muted)] mt-0.5">Switch the entire interface to RTL for Arabic/Hebrew languages</div>
   339→                </div>
   340→                <button
   341→                  onClick={() => { setRtl(!rtl); update('rtl', String(!rtl)); }}
   342→                  className={cn(
   343→                    'relative w-11 h-6 rounded-full transition-colors shrink-0',
   344→                    rtl ? 'bg-[var(--erp-accent)]' : 'bg-[var(--erp-bg-hover)]',
   345→                  )}
   346→                  aria-label="Toggle RTL"
   347→                >
   348→                  <span className={cn(
   349→                    'absolute top-0.5 w-5 h-5 rounded-full bg-white transition-transform',
   350→                    rtl ? 'translate-x-5' : 'translate-x-0.5',
   351→                  )} />
   352→                </button>
   353→                <span className={cn('text-[11px] font-medium', rtl ? 'text-[var(--erp-accent)]' : 'text-[var(--erp-text-muted)]')}>
   354→                  {rtl ? 'RTL' : 'LTR'}
   355→                </span>
   356→              </div>
   357→            </div>
   358→          )}
   359→
   360→          {activeTab === 'numbering' && (
   361→            <div className="space-y-4 max-w-2xl">
   362→              <h2 className="text-[15px] font-semibold text-[var(--erp-text)] flex items-center gap-2">
   363→                <Hash className="w-4 h-4 text-[var(--erp-accent)]" /> Document Numbering
   364→              </h2>
   365→              <p className="text-[12px] text-[var(--erp-text-muted)]">
   366→                Document numbers are auto-generated from sequence counters. The prefix is derived from the column name (e.g. "WO Number" → WO-0001).
   367→                This is a system-managed feature — no manual configuration needed.

   371→                <div className="grid grid-cols-2 sm:grid-cols-3 gap-2 text-[12px]">
   372→                  {['WO-0001', 'PM-0001', 'PTW-0001', 'INV-0001', 'AST-0001', 'VND-0001'].map((n) => (
   373→                    <div key={n} className="font-mono text-[var(--erp-accent)] bg-[var(--erp-bg-input)] px-2 py-1 rounded">{n}</div>
   374→                  ))}
   375→                </div>
   376→              </div>
   377→            </div>
   378→          )}
   379→
   380→          {activeTab === 'notifications' && (
   381→            <div className="space-y-4 max-w-2xl">
   382→              <h2 className="text-[15px] font-semibold text-[var(--erp-text)] flex items-center gap-2">
   383→                <Bell className="w-4 h-4 text-[var(--erp-accent)]" /> Email Notifications
   384→              </h2>
   385→              <p className="text-[12px] text-[var(--erp-text-muted)]">

   387→                In production, configure Resend or SendGrid for actual delivery.
   388→              </p>
   389→
   390→              <div className="bg-[var(--erp-bg-card)] border border-[var(--erp-border)] rounded-md p-4 space-y-3">
   391→                <div className="text-[12px] font-semibold text-[var(--erp-text)]">Email Notification Triggers</div>

   397→                  { key: 'email.on_pm_due', label: 'PM Due Reminder', desc: 'When preventive maintenance is due', default: 'false' },

   400→                  { key: 'email.on_weekly_summary', label: 'Weekly Summary', desc: 'Weekly summary every Monday at 8:00 AM', default: 'false' },
   401→                ].map((item) => (
   402→                  <div key={item.key} className="flex items-center justify-between py-1.5 border-b border-[var(--erp-border)] last:border-0">
   403→                    <div className="flex-1">
   404→                      <div className="text-[12px] font-medium text-[var(--erp-text)]">{item.label}</div>

   406→                    </div>

   409→                      defaultChecked={settings[item.key] === 'true' || (item.default === 'true' && !settings[item.key])}
